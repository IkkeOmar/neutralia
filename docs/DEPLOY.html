Deploy checklist (run every time)

export GH_TOKEN="$SKILLS_HUB_GITHUB_TOKEN"
WORK=/home/omar/2026-11-07/bfh-mmo
PUB=$WORK/_pages                      # staging tree, deleted after deploy

# 1. rebuild the staging tree
rm -rf "$PUB" && mkdir -p "$PUB/prototype" "$PUB/docs"
cp "$WORK"/prototype/*.html "$PUB/prototype/"
cp "$WORK"/prototype/*.js   "$PUB/prototype/" 2>/dev/null || true
python3 "$WORK"/tools/render-docs.py        # markdown -> docs/*.html

# 2. push
cd "$PUB"
git init -q 2>/dev/null || true
git remote set-url origin "https://x-access-token:$SKILLS_HUB_GITHUB_TOKEN@github.com/IkkeOmar/neutralia.git"
git add -A
git -c commit.gpgsign=false commit -q -m "<message>" || true
git push -q origin main

# 3. wait for the Pages build — the CDN serves the PREVIOUS build for ~30-60s
for i in $(seq 1 12); do
  s=$(curl -s -H "Authorization: Bearer $GH_TOKEN" \
      https://api.github.com/repos/IkkeOmar/neutralia/pages \
      | python3 -c "import json,sys; print(json.load(sys.stdin).get('status'))")
  [ "$s" = "built" ] && break
  sleep 10
done

# 4. verify against the LIVE url, cache-busted
curl -s "https://ikkeomar.github.io/neutralia/prototype/classes.html?cb=$(date +%s)" \
  | grep -q "New</span>" && echo "live OK"

# 5. clean up
rm -rf "$PUB"

Stale-CDN trap: curl the live URL too early and you get the old build. Poll Pages status to built, then fetch with ?cb=$(date +%s) before concluding a change did not land.
